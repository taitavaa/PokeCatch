# pokemon_dex_fetcher.py

import requests

def fetch_pokemon_dex_entries():
    """
    Fetches every Pokémon dex entry using the PokéAPI.
    """
    # Fetch the list of all Pokémon
    response = requests.get('https://pokeapi.co/api/v2/pokemon?limit=1000', headers={'Accept': 'application/json'})
    pokemon_list = response.json()['results']

    # Iterate through the list and fetch each Pokémon's dex entry
    for pokemon in pokemon_list:
        pokemon_name = pokemon['name']
        response = requests.get(f'https://pokeapi.co/api/v2/pokemon/{pokemon_name}', headers={'Accept': 'application/json'})
        pokemon_data = response.json()
        print(f"Species data for {pokemon_name}: {pokemon_data['species']}")
        # dex_entry = pokemon_data['species']['flavor_text_entries'][0]['flavor_text']
        # print(f"{pokemon_name}: {dex_entry}")

if __name__ == '__main__':
    fetch_pokemon_dex_entries()
